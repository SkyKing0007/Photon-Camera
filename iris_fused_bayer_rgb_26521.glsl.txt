#version 300 es
precision highp float;
precision highp sampler2D;

uniform sampler2D fusedCfa;
uniform sampler2D lensShadingMap;
uniform ivec2 rawSize;
uniform ivec2 packedSize;
uniform int cfaPattern;
uniform int useLensShading;

layout(location = 0) out vec4 Output;

/*
 * IRIS_26521_INDEPENDENT_FUSED_BAYER_RGB
 *
 * Independently authored Iris RGB reconstruction.
 *
 * Input is the already-aligned, robustness-weighted NORMAL Bayer accumulator,
 * after optional separate Short/Long Bayer-domain HDR recovery for JPEG only.
 * This shader does not use the released c4ff Spatial RGB accumulator, its
 * opponent-color representation, covariance kernel, constants, or shaders.
 *
 * Reconstruction:
 *  1. preserve measured green on green sites;
 *  2. estimate green on red/blue sites from horizontal/vertical gradients;
 *  3. reconstruct missing red/blue from local color-minus-green residuals;
 *  4. optionally apply Camera2 lens-shading gains in camera RGB space.
 *
 * Extended-linear values above 1.0 are preserved.
 */

int phaseAt(ivec2 p) {
    return ((p.y & 1) << 1) | (p.x & 1);
}

int colorForPhase(int phase) {
    if (cfaPattern == 0) {
        if (phase == 0) return 0;
        if (phase == 3) return 2;
        return 1;
    }
    if (cfaPattern == 1) {
        if (phase == 1) return 0;
        if (phase == 2) return 2;
        return 1;
    }
    if (cfaPattern == 2) {
        if (phase == 2) return 0;
        if (phase == 1) return 2;
        return 1;
    }
    if (phase == 3) return 0;
    if (phase == 0) return 2;
    return 1;
}

int firstPhaseForColor(int color) {
    for (int phase = 0; phase < 4; ++phase) {
        if (colorForPhase(phase) == color) return phase;
    }
    return 0;
}

float packedPhase(vec4 v, int phase) {
    if (phase == 0) return v.r;
    if (phase == 1) return v.g;
    if (phase == 2) return v.b;
    return v.a;
}

float rawAt(ivec2 p) {
    p = clamp(p, ivec2(0), rawSize - ivec2(1));
    ivec2 q = clamp(p / 2, ivec2(0), packedSize - ivec2(1));
    vec4 packed = texelFetch(fusedCfa, q, 0);
    return max(packedPhase(packed, phaseAt(p)), 0.0);
}

float greenAt(ivec2 p) {
    int ownColor = colorForPhase(phaseAt(p));
    float center = rawAt(p);
    if (ownColor == 1) return center;

    float left  = rawAt(p + ivec2(-1, 0));
    float right = rawAt(p + ivec2( 1, 0));
    float up    = rawAt(p + ivec2( 0,-1));
    float down  = rawAt(p + ivec2( 0, 1));

    float sameLeft  = rawAt(p + ivec2(-2, 0));
    float sameRight = rawAt(p + ivec2( 2, 0));
    float sameUp    = rawAt(p + ivec2( 0,-2));
    float sameDown  = rawAt(p + ivec2( 0, 2));

    float horizontal = 0.5 * (left + right);
    float vertical   = 0.5 * (up + down);

    float gradH = abs(left - right)
            + 0.5 * abs(2.0 * center - sameLeft - sameRight);
    float gradV = abs(up - down)
            + 0.5 * abs(2.0 * center - sameUp - sameDown);

    float weightH = 1.0 / (1.0 + 18.0 * gradH * gradH);
    float weightV = 1.0 / (1.0 + 18.0 * gradV * gradV);
    return max((horizontal * weightH + vertical * weightV)
            / max(weightH + weightV, 1.0e-6), 0.0);
}

float reconstructColor(ivec2 p, int wantedColor, float targetGreen) {
    int ownColor = colorForPhase(phaseAt(p));
    if (ownColor == wantedColor) return rawAt(p);
    if (wantedColor == 1) return targetGreen;

    float residualSum = 0.0;
    float weightSum = 0.0;
    for (int dy = -1; dy <= 1; ++dy) {
        for (int dx = -1; dx <= 1; ++dx) {
            if (dx == 0 && dy == 0) continue;
            ivec2 q = clamp(p + ivec2(dx, dy),
                            ivec2(0), rawSize - ivec2(1));
            if (colorForPhase(phaseAt(q)) != wantedColor) continue;

            float gq = greenAt(q);
            float residual = rawAt(q) - gq;
            float distance2 = float(dx * dx + dy * dy);
            float spatial = 1.0 / (1.0 + distance2);
            float edgeAgreement =
                    1.0 / (1.0 + 14.0 * abs(gq - targetGreen));
            float w = spatial * edgeAgreement;
            residualSum += residual * w;
            weightSum += w;
        }
    }
    if (weightSum <= 1.0e-6) return targetGreen;
    return max(targetGreen + residualSum / weightSum, 0.0);
}

vec3 lensShadingRgb(vec2 uv) {
    if (useLensShading == 0) return vec3(1.0);
    vec4 p = texture(lensShadingMap, uv);
    int rp = firstPhaseForColor(0);
    int bp = firstPhaseForColor(2);
    float r = packedPhase(p, rp);
    float b = packedPhase(p, bp);
    float gsum = 0.0;
    float gcount = 0.0;
    for (int phase = 0; phase < 4; ++phase) {
        if (colorForPhase(phase) == 1) {
            gsum += packedPhase(p, phase);
            gcount += 1.0;
        }
    }
    float g = gcount > 0.0 ? gsum / gcount : 1.0;
    return max(vec3(r, g, b), vec3(0.0));
}

void main() {
    ivec2 p = clamp(ivec2(gl_FragCoord.xy),
                    ivec2(0), rawSize - ivec2(1));
    float g = greenAt(p);
    float r = reconstructColor(p, 0, g);
    float b = reconstructColor(p, 2, g);
    vec2 uv = (vec2(p) + vec2(0.5)) / vec2(rawSize);
    vec3 rgb = vec3(r, g, b) * lensShadingRgb(uv);
    Output = vec4(max(rgb, vec3(0.0)), 1.0);
}
